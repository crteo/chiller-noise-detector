#include <math.h>

// ======================================================
// WIFI
// ======================================================

const char* ssid = "Teo Family";
const char* password = "";

// ======================================================
// FIREBASE
// ======================================================

const char* firebaseURL =
  "https://science-park-noise-monitoring-default-rtdb.asia-southeast1.firebasedatabase.app/current.json";

// ======================================================
// INMP441 / I2S PINS
// ======================================================

#define I2S_BCLK   18
#define I2S_LRCLK  16
#define I2S_DIN    36

#define I2S_PORT I2S_NUM_0

// ======================================================
// AUDIO SETTINGS
// ======================================================

const uint32_t SAMPLE_RATE = 48000;
const uint16_t FFT_SIZE = 4096;

// Nominal INMP441 sensitivity-based conversion
const double CALIB_CONST = 123.01;

// 12 × 4096 = 49152 samples
const uint8_t FRAMES_PER_LEVEL = 12;

const uint32_t LEVEL_SAMPLE_COUNT =
  FFT_SIZE * FRAMES_PER_LEVEL;

const double LEVEL_DURATION_SECONDS =
  (double)LEVEL_SAMPLE_COUNT /
  SAMPLE_RATE;

// 24-bit signed PCM
const double PCM_FULL_SCALE =
  8388608.0; // 2^23

// ======================================================
// I2S BUFFER
// ======================================================

const int I2S_BUFFER_SIZE = 512;

int32_t i2sBuffer[I2S_BUFFER_SIZE];

// ======================================================
// FFT ARRAYS
// ======================================================

double vReal[FFT_SIZE];
double vImag[FFT_SIZE];

ArduinoFFT<double> FFT(
  vReal,
  vImag,
  FFT_SIZE,
  SAMPLE_RATE
);

uint16_t fftIndex = 0;

// ======================================================
// LONGER-TERM ENERGY ACCUMULATORS
// ======================================================

double accumulatedUnweightedMS = 0.0;
double accumulatedWeightedMS = 0.0;

double accumulatedDominantFrequency = 0.0;

uint8_t accumulatedFrames = 0;

// ======================================================
// SHARED RESULTS
// ======================================================

SemaphoreHandle_t resultMutex;

double latestDBFS = -120.0;
double latestDBFSA = -120.0;

double latestRMS = 0.0;
double latestRMSA = 0.0;

double latestACorrection = 0.0;

double latestDominantFrequency = 0.0;

double latestFFTError = 0.0;

double latestEstimatedDBA = 0.0;

unsigned long latestResultID = 0;

// ======================================================
// A-WEIGHTING EQUATION
// ======================================================

double calculateAWeightingDB(double f)
{
  if (f <= 0.0)
  {
    return -200.0;
  }

  const double c1 = 20.6;
  const double c2 = 107.7;
  const double c3 = 737.9;
  const double c4 = 12194.0;

  double f2 = f * f;

  double c1_2 = c1 * c1;
  double c2_2 = c2 * c2;
  double c3_2 = c3 * c3;
  double c4_2 = c4 * c4;

  double numerator =
    c4_2 *
    f2 *
    f2;

  double denominator =
    (f2 + c1_2) *
    sqrt(
      (f2 + c2_2) *
      (f2 + c3_2)
    ) *
    (f2 + c4_2);

  double RA =
    numerator /
    denominator;

  if (RA <= 0.0)
  {
    return -200.0;
  }

  return
    20.0 * log10(RA)
    + 2.0;
}

// ======================================================
// A-WEIGHTING POWER GAIN
// ======================================================

double calculateAWeightPowerGain(double frequency)
{
  double weightingDB =
    calculateAWeightingDB(
      frequency
    );

  return pow(
    10.0,
    weightingDB / 10.0
  );
}

// ======================================================
// WIFI
// ======================================================

void setupWiFi()
{
  Serial.println();
  Serial.print("Connecting to Wi-Fi");

  WiFi.mode(WIFI_STA);

  WiFi.begin(
    ssid,
    password
  );

  while (
    WiFi.status() != WL_CONNECTED
  )
  {
    delay(500);
    Serial.print(".");
  }

  Serial.println();
  Serial.println("Wi-Fi connected.");

  Serial.print("IP: ");
  Serial.println(WiFi.localIP());
}

// ======================================================
// I2S
// ======================================================

void setupI2S()
{
  i2s_config_t i2sConfig = {

    .mode =
      (i2s_mode_t)(
        I2S_MODE_MASTER |
        I2S_MODE_RX
      ),

    .sample_rate =
      SAMPLE_RATE,

    .bits_per_sample =
      I2S_BITS_PER_SAMPLE_32BIT,

    // Assumes L/R pin tied to GND
    .channel_format =
      I2S_CHANNEL_FMT_ONLY_LEFT,

    .communication_format =
      I2S_COMM_FORMAT_STAND_I2S,

    .intr_alloc_flags =
      ESP_INTR_FLAG_LEVEL1,

    .dma_buf_count =
      16,

    .dma_buf_len =
      256,

    .use_apll =
      false,

    .tx_desc_auto_clear =
      false,

    .fixed_mclk =
      0
  };

  i2s_pin_config_t pinConfig = {

    .bck_io_num =
      I2S_BCLK,

    .ws_io_num =
      I2S_LRCLK,

    .data_out_num =
      I2S_PIN_NO_CHANGE,

    .data_in_num =
      I2S_DIN
  };

  esp_err_t result =
    i2s_driver_install(
      I2S_PORT,
      &i2sConfig,
      0,
      NULL
    );

  if (result != ESP_OK)
  {
    Serial.print(
      "I2S install failed: "
    );

    Serial.println(result);

    while (true)
    {
      delay(1000);
    }
  }

  result =
    i2s_set_pin(
      I2S_PORT,
      &pinConfig
    );

  if (result != ESP_OK)
  {
    Serial.print(
      "I2S pin configuration failed: "
    );

    Serial.println(result);

    while (true)
    {
      delay(1000);
    }
  }

  Serial.println("I2S initialized.");
}

// ======================================================
// FINALISE ~1.024 SECOND LEVEL
// ======================================================

void finishLevelMeasurement()
{
  if (
    accumulatedFrames == 0
  )
  {
    return;
  }

  // ----------------------------------------------------
  // Energy averaging
  // ----------------------------------------------------

  double averageMS =
    accumulatedUnweightedMS /
    accumulatedFrames;

  double averageMSA =
    accumulatedWeightedMS /
    accumulatedFrames;

  if (averageMS < 0.0)
  {
    averageMS = 0.0;
  }

  if (averageMSA < 0.0)
  {
    averageMSA = 0.0;
  }

  double rms =
    sqrt(
      averageMS
    );

  double rmsA =
    sqrt(
      averageMSA
    );

  // ----------------------------------------------------
  // dBFS
  // ----------------------------------------------------

  double dbfs =
    -120.0;

  if (rms > 1e-12)
  {
    dbfs =
      20.0 *
      log10(rms);
  }

  double dbfsA =
    -120.0;

  if (rmsA > 1e-12)
  {
    dbfsA =
      20.0 *
      log10(rmsA);
  }

  double aCorrection =
    dbfsA -
    dbfs;

  double dominantFrequency =
    accumulatedDominantFrequency /
    accumulatedFrames;

  // ----------------------------------------------------
  // Estimated dBA
  // ----------------------------------------------------

  double estimatedDBA =
    dbfsA +
    CALIB_CONST;

  // ====================================================
  // SERIAL OUTPUT
  // ====================================================

  Serial.println();
  Serial.println(
    "=========================================="
  );

  Serial.println(
    "       INTEGRATED LEVEL RESULT"
  );

  Serial.println(
    "=========================================="
  );

  Serial.print(
    "FFT frames              : "
  );

  Serial.println(
    accumulatedFrames
  );

  Serial.print(
    "Samples                 : "
  );

  Serial.println(
    LEVEL_SAMPLE_COUNT
  );

  Serial.print(
    "Integration duration    : "
  );

  Serial.print(
    LEVEL_DURATION_SECONDS,
    3
  );

  Serial.println(
    " s"
  );

  Serial.println(
    "------------------------------------------"
  );

  Serial.print(
    "Unweighted RMS          : "
  );

  Serial.println(
    rms,
    8
  );

  Serial.print(
    "Unweighted level        : "
  );

  Serial.print(
    dbfs,
    2
  );

  Serial.println(
    " dBFS"
  );

  Serial.print(
    "A-weighted RMS          : "
  );

  Serial.println(
    rmsA,
    8
  );

  Serial.print(
    "A-weighted level        : "
  );

  Serial.print(
    dbfsA,
    2
  );

  Serial.println(
    " dBFS(A)"
  );

  Serial.print(
    "Estimated A-weighted SPL: "
  );

  Serial.print(
    estimatedDBA,
    2
  );

  Serial.println(
    " dBA"
  );

  Serial.print(
    "A-weight difference     : "
  );

  Serial.print(
    aCorrection,
    2
  );

  Serial.println(
    " dB"
  );

  Serial.println(
    "------------------------------------------"
  );

  Serial.print(
    "Avg dominant frequency  : "
  );

  Serial.print(
    dominantFrequency,
    2
  );

  Serial.println(
    " Hz"
  );

  Serial.println(
    "=========================================="
  );

  // ====================================================
  // SHARE RESULT WITH FIREBASE TASK
  // ====================================================

  if (
    xSemaphoreTake(
      resultMutex,
      portMAX_DELAY
    ) == pdTRUE
  )
  {
    latestDBFS =
      dbfs;

    latestDBFSA =
      dbfsA;

    latestRMS =
      rms;

    latestRMSA =
      rmsA;

    latestACorrection =
      aCorrection;

    latestDominantFrequency =
      dominantFrequency;

    latestEstimatedDBA =
      estimatedDBA;

    latestResultID++;

    xSemaphoreGive(
      resultMutex
    );
  }

  // ====================================================
  // RESET INTEGRATION
  // ====================================================

  accumulatedUnweightedMS =
    0.0;

  accumulatedWeightedMS =
    0.0;

  accumulatedDominantFrequency =
    0.0;

  accumulatedFrames =
    0;
}

// ======================================================
// PROCESS ONE FFT FRAME
// ======================================================

void processFFTFrame()
{
  // ----------------------------------------------------
  // 1. Remove mean / DC
  // ----------------------------------------------------

  double mean =
    0.0;

  for (
    uint16_t i = 0;
    i < FFT_SIZE;
    i++
  )
  {
    mean +=
      vReal[i];
  }

  mean /=
    FFT_SIZE;

  for (
    uint16_t i = 0;
    i < FFT_SIZE;
    i++
  )
  {
    vReal[i] -=
      mean;
  }

  // ----------------------------------------------------
  // 2. Hann window
  // ----------------------------------------------------

  double windowSquareSum =
    0.0;

  double windowedSquareSum =
    0.0;

  for (
    uint16_t i = 0;
    i < FFT_SIZE;
    i++
  )
  {
    double w =
      0.5 *
      (
        1.0 -
        cos(
          2.0 *
          PI *
          i /
          (FFT_SIZE - 1)
        )
      );

    vReal[i] *=
      w;

    vImag[i] =
      0.0;

    windowSquareSum +=
      w * w;

    windowedSquareSum +=
      vReal[i] *
      vReal[i];
  }

  double U =
    windowSquareSum /
    FFT_SIZE;

  // ----------------------------------------------------
  // 3. Time-domain QA reference
  // ----------------------------------------------------

  double timeMeanSquare =
    windowedSquareSum /
    (
      FFT_SIZE *
      U
    );

  double timeRMS =
    sqrt(
      timeMeanSquare
    );

  double timeDBFS =
    -120.0;

  if (timeRMS > 1e-12)
  {
    timeDBFS =
      20.0 *
      log10(timeRMS);
  }

  // ----------------------------------------------------
  // 4. FFT
  // ----------------------------------------------------

  FFT.compute(
    FFTDirection::Forward
  );

  // ----------------------------------------------------
  // 5. Spectral energy
  // ----------------------------------------------------

  double spectralEnergy =
    0.0;

  double weightedEnergy =
    0.0;

  double maxBinEnergy =
    0.0;

  uint16_t maxBin =
    0;

  for (
    uint16_t k = 0;
    k <= FFT_SIZE / 2;
    k++
  )
  {
    double re =
      vReal[k];

    double im =
      vImag[k];

    double magnitudeSquared =
      re * re +
      im * im;

    double binEnergy =
      magnitudeSquared;

    // One-sided spectrum energy correction
    if (
      k > 0 &&
      k < FFT_SIZE / 2
    )
    {
      binEnergy *=
        2.0;
    }

    spectralEnergy +=
      binEnergy;

    // --------------------------------------------------
    // A-weighting
    // --------------------------------------------------

    if (k > 0)
    {
      double frequency =
        (
          (double)k *
          SAMPLE_RATE
        ) /
        FFT_SIZE;

      double weight =
        calculateAWeightPowerGain(
          frequency
        );

      weightedEnergy +=
        binEnergy *
        weight;
    }

    // --------------------------------------------------
    // Dominant frequency
    // --------------------------------------------------

    if (
      k > 0 &&
      k < FFT_SIZE / 2 &&
      binEnergy > maxBinEnergy
    )
    {
      maxBinEnergy =
        binEnergy;

      maxBin =
        k;
    }
  }

  // ----------------------------------------------------
  // 6. Parseval / Hann normalization
  // ----------------------------------------------------

  double normalization =
    (
      (double)FFT_SIZE *
      (double)FFT_SIZE *
      U
    );

  double fftMeanSquare =
    spectralEnergy /
    normalization;

  double weightedMeanSquare =
    weightedEnergy /
    normalization;

  if (fftMeanSquare < 0.0)
  {
    fftMeanSquare = 0.0;
  }

  if (weightedMeanSquare < 0.0)
  {
    weightedMeanSquare = 0.0;
  }

  // ----------------------------------------------------
  // 7. FFT QA check
  // ----------------------------------------------------

  double fftRMS =
    sqrt(
      fftMeanSquare
    );

  double fftDBFS =
    -120.0;

  if (fftRMS > 1e-12)
  {
    fftDBFS =
      20.0 *
      log10(fftRMS);
  }

  double fftError =
    fftDBFS -
    timeDBFS;

  double dominantFrequency =
    (
      (double)maxBin *
      SAMPLE_RATE
    ) /
    FFT_SIZE;

  // ----------------------------------------------------
  // Diagnostics
  // ----------------------------------------------------

  Serial.print(
    "[FFT] f_dom="
  );

  Serial.print(
    dominantFrequency,
    1
  );

  Serial.print(
    " Hz | FFT error="
  );

  Serial.print(
    fftError,
    4
  );

  Serial.print(
    " dB | frame "
  );

  Serial.print(
    accumulatedFrames + 1
  );

  Serial.print(
    "/"
  );

  Serial.println(
    FRAMES_PER_LEVEL
  );

  // ----------------------------------------------------
  // 8. Energy accumulation
  // ----------------------------------------------------

  accumulatedUnweightedMS +=
    fftMeanSquare;

  accumulatedWeightedMS +=
    weightedMeanSquare;

  accumulatedDominantFrequency +=
    dominantFrequency;

  accumulatedFrames++;

  // Keep latest FFT QA value available
  if (
    xSemaphoreTake(
      resultMutex,
      portMAX_DELAY
    ) == pdTRUE
  )
  {
    latestFFTError =
      fftError;

    xSemaphoreGive(
      resultMutex
    );
  }

  // ----------------------------------------------------
  // 12 FFT frames complete
  // ----------------------------------------------------

  if (
    accumulatedFrames >=
    FRAMES_PER_LEVEL
  )
  {
    finishLevelMeasurement();
  }
}

// ======================================================
// AUDIO TASK
// ======================================================

void audioTask(
  void* parameter
)
{
  while (true)
  {
    size_t bytesRead =
      0;

    esp_err_t result =
      i2s_read(
        I2S_PORT,
        i2sBuffer,
        sizeof(i2sBuffer),
        &bytesRead,
        portMAX_DELAY
      );

    if (
      result != ESP_OK
    )
    {
      Serial.println(
        "[AUDIO] I2S read error."
      );

      continue;
    }

    int samplesReceived =
      bytesRead /
      sizeof(int32_t);

    for (
      int i = 0;
      i < samplesReceived;
      i++
    )
    {
      int32_t pcm =
        i2sBuffer[i] >> 8;

      double normalized =
        (double)pcm /
        PCM_FULL_SCALE;

      vReal[fftIndex] =
        normalized;

      vImag[fftIndex] =
        0.0;

      fftIndex++;

      if (
        fftIndex >= FFT_SIZE
      )
      {
        processFFTFrame();

        fftIndex =
          0;
      }
    }
  }
}

// ======================================================
// FIREBASE UPLOAD
// ======================================================

void uploadFirebase(
  double dbfs,
  double dbfsA,
  double rms,
  double rmsA,
  double correction,
  double dominantFrequency,
  double fftError,
  double estimatedDBA
)
{
  if (
    WiFi.status() != WL_CONNECTED
  )
  {
    Serial.println(
      "[UPLOAD] Wi-Fi disconnected."
    );

    return;
  }

  WiFiClientSecure client;

  // Prototype only
  client.setInsecure();

  HTTPClient https;

  if (
    !https.begin(
      client,
      firebaseURL
    )
  )
  {
    Serial.println(
      "[UPLOAD] HTTPS begin failed."
    );

    return;
  }

  https.addHeader(
    "Content-Type",
    "application/json"
  );

  String json = "{";

  json +=
    "\"dbfs\":" +
    String(dbfs, 2);

  json +=
    ",\"dbfsA\":" +
    String(dbfsA, 2);

  json +=
    ",\"estimated_dba\":" +
    String(estimatedDBA, 2);

  json +=
    ",\"rms\":" +
    String(rms, 8);

  json +=
    ",\"rmsA\":" +
    String(rmsA, 8);

  json +=
    ",\"a_correction\":" +
    String(correction, 2);

  json +=
    ",\"dominant_frequency\":" +
    String(
      dominantFrequency,
      2
    );

  json +=
    ",\"fft_error_db\":" +
    String(
      fftError,
      4
    );

  json +=
    ",\"fft_size\":" +
    String(
      FFT_SIZE
    );

  json +=
    ",\"frames_per_level\":" +
    String(
      FRAMES_PER_LEVEL
    );

  json +=
    ",\"measurement_samples\":" +
    String(
      LEVEL_SAMPLE_COUNT
    );

  json +=
    ",\"measurement_duration_s\":" +
    String(
      LEVEL_DURATION_SECONDS,
      3
    );

  json +=
    ",\"sample_rate\":" +
    String(
      SAMPLE_RATE
    );

  json +=
    ",\"uptime_ms\":" +
    String(
      millis()
    );

  json += "}";

  Serial.print(
    "[UPLOAD] estimated_dba="
  );

  Serial.println(
    estimatedDBA,
    2
  );

  unsigned long start =
    millis();

  int httpCode =
    https.PUT(
      json
    );

  unsigned long duration =
    millis() -
    start;

  Serial.print(
    "[UPLOAD] HTTP "
  );

  Serial.print(
    httpCode
  );

  Serial.print(
    " | "
  );

  Serial.print(
    duration
  );

  Serial.println(
    " ms"
  );

  https.end();
}

// ======================================================
// FIREBASE TASK
// ======================================================

void firebaseTask(
  void* parameter
)
{
  unsigned long previousResultID =
    0;

  while (true)
  {
    double dbfs;
    double dbfsA;

    double rms;
    double rmsA;

    double correction;

    double dominantFrequency;

    double fftError;

    double estimatedDBA;

    unsigned long resultID;

    // --------------------------------------------------
    // Copy shared results
    // --------------------------------------------------

    if (
      xSemaphoreTake(
        resultMutex,
        portMAX_DELAY
      ) == pdTRUE
    )
    {
      dbfs =
        latestDBFS;

      dbfsA =
        latestDBFSA;

      rms =
        latestRMS;

      rmsA =
        latestRMSA;

      correction =
        latestACorrection;

      dominantFrequency =
        latestDominantFrequency;

      fftError =
        latestFFTError;

      estimatedDBA =
        latestEstimatedDBA;

      resultID =
        latestResultID;

      xSemaphoreGive(
        resultMutex
      );
    }

    // --------------------------------------------------
    // Upload only new integrated result
    // --------------------------------------------------

    if (
      resultID != 0 &&
      resultID != previousResultID
    )
    {
      uploadFirebase(
        dbfs,
        dbfsA,
        rms,
        rmsA,
        correction,
        dominantFrequency,
        fftError,
        estimatedDBA
      );

      previousResultID =
        resultID;
    }

    vTaskDelay(
      pdMS_TO_TICKS(100)
    );
  }
}

// ======================================================
// SETUP
// ======================================================

void setup()
{
  Serial.begin(
    115200
  );

  delay(
    1500
  );

  Serial.println();
  Serial.println(
    "=========================================="
  );

  Serial.println(
    "INMP441 FFT A-WEIGHTED LEVEL MONITOR"
  );

  Serial.print(
    "FFT size: "
  );

  Serial.println(
    FFT_SIZE
  );

  Serial.print(
    "Frames per result: "
  );

  Serial.println(
    FRAMES_PER_LEVEL
  );

  Serial.print(
    "Actual integration time: "
  );

  Serial.print(
    LEVEL_DURATION_SECONDS,
    3
  );

  Serial.println(
    " s"
  );

  Serial.print(
    "Nominal dBA calibration constant: "
  );

  Serial.println(
    CALIB_CONST,
    2
  );

  Serial.println(
    "=========================================="
  );

  resultMutex =
    xSemaphoreCreateMutex();

  if (
    resultMutex == NULL
  )
  {
    Serial.println(
      "ERROR: Mutex creation failed."
    );

    while (true)
    {
      delay(1000);
    }
  }

  setupI2S();

  setupWiFi();

  // Audio task
  xTaskCreatePinnedToCore(
    audioTask,
    "AudioTask",
    12288,
    NULL,
    2,
    NULL,
    1
  );

  // Firebase task
  xTaskCreatePinnedToCore(
    firebaseTask,
    "FirebaseTask",
    12288,
    NULL,
    1,
    NULL,
    0
  );

  Serial.println();
  Serial.println(
    "System ready."
  );
}

// ======================================================
// LOOP
// ======================================================

void loop()
{
  delay(
    1000
  );
}
